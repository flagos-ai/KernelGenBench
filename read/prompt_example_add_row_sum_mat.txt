import triton
import triton.language as tl
import torch

@triton.jit
def add_kernel(mat_ptr,  # *Pointer* to input matrix.
            value,  # Scalar value to add.
            num_rows,  # Number of rows in the matrix.
            num_cols,  # Number of columns in the matrix.
            row_stride,  # Stride between rows.
            BLOCK_SIZE: tl.constexpr,  # Number of elements each program should process.
            # NOTE: `constexpr` so it can be used as a shape value.
            ):
    pid = tl.program_id(axis=0)  # We use a 1D launch grid so axis is 0.
    if pid >= num_rows:
        return
    row_start = pid * row_stride
    col_offsets = tl.arange(0, BLOCK_SIZE)
    offsets = row_start + col_offsets
    mask = col_offsets < num_cols
    mat = tl.load(mat_ptr + offsets, mask=mask)
    result = mat + value
    tl.store(mat_ptr + offsets, result, mask=mask)

def add(mat: torch.Tensor, value: float):
    assert mat.is_cuda, "Input tensor must be on CUDA"
    num_rows, num_cols = mat.shape
    row_stride = mat.stride(0)
    grid = lambda meta: (num_rows,)
    add_kernel[grid](mat, value, num_rows, num_cols, row_stride, BLOCK_SIZE=1024)
    return mat
